import turtle import math import time screen = turtle.Screen() screen.setup(900, 800) screen.bgcolor("black") screen.title("Seni Her Dilde Seviyorum") screen.tracer(0) t = turtle.Turtle() t.hideturtle() t.speed(0) t.pensize(1) pinks = ["#ffb6d9", "#ff69b4", "#ff8fc8", "#ffc0e0", "#ff4fa3"] def heart(a): x = 16 * math.sin(a) ** 3 y = (13 * math.cos(a) - 5 * math.cos(2*a) - 2 * math.cos(3*a) - math.cos(4*a)) return x, y # Kalbi ciz for i in range(180): a = 2 * math.pi * i / 180 x, y = heart(a) t.pencolor(pinks[i % len(pinks)]) t.penup() t.goto(x * 13, y * 13) t.pendown() t.goto(x * 5, y * 5) screen.update() time.sleep(0.01) # Farkli dillerde seni seviyorum words = [ "Seni seviyorum", "I love you", "Je t'aime", "Te amo", "Ti amo", "Ich liebe dich", "Eu te amo", "愛してる", "사랑해", "我爱你", "Я тебя люблю", "Kocham cię", "Szeretlek", "Ik hou van jou", "Jag älskar dig", "Mahal kita", "Nakupenda", "Volim te", "Te iubesc", "S'agapo", "Seni sevirem", "Amo-te" ] w = turtle.Turtle() w.hideturtle() w.penup() w.speed(0) for i, word in enumerate(words): a = 2 * math.pi * i / len(words) x = 275 * math.cos(a) y = 285 * math.sin(a) w.goto(x, y) w.color(pinks[i % len(pinks)]) w.write(word, align="center", font=("Arial", 9, "normal")) screen.update() # Ortadaki mesaj w.goto(0, -20) w.color("#ffd6e9") w.write("SENI SEVIYORUM", align="center", font=("Arial", 15, "bold")) screen.update() turtle.done()
